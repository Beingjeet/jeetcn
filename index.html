#include stdio.h
#include string.h
int main(){
char data[100];
int i, count = 0;
char parity;
// Taking the bits
printf("Enter the bits (0s and 1s only): ");
scanf("%s", data);
printf("Enter Parity type (O for Odd, E for Even): ");
scanf(" %c", &parity);
// Calculate parity
for(i = 0; data[i] != '\0'; i++){
if(data[i] == '1')
count++;
}
int len = strlen(data);
if(parity == 'O' || parity == 'o') {
if(count % 2 == 0)
data[len] = '1';
else
data[len] = '0';
data[len + 1] = '\0';
printf("The data with parity bit is: %s\n", data);
}
else if(parity == 'E' || parity == 'e') {
if(count % 2 == 0)
data[len] = '0';
else
data[len] = '1';
data[len + 1] = '\0';
printf("The data with parity bit is: %s\n", data);
}
else {
printf("Invalid parity type entered.\n");
}
return 0;
}

Dijkstra's Algorithm

int main() {
int V = 7;
int graph[7][7] = {
{0, 2, 6, 0, 0, 0, 0},
{2, 0, 0, 5, 0, 0, 0},
{6, 0, 0, 8, 0, 0, 0},
{0, 5, 8, 0,10,15, 0},
{0, 0, 0,10, 0, 0, 2},
{0, 0, 0,15, 0, 0, 6},
{0, 0, 0, 0, 2, 6, 0}
};
int dist[V],
visited[V] = {0};
// initialize distances
for (int i = 0; i < V; i++)
dist[i] = INT_MAX;
// starting from vertex 0
dist[0] = 0;
for (int i = 0; i < V; i++) {
int u = -1;
for (int j = 0; j < V; j++)
if (!visited[j] && (u == -1 || dist[j] < dist[u]))
u = j; }
//Mark vertex as visited
visited[u] = 1;
//Update distance of neighbouring vertices
for (int v = 0; v < V; v++)
if (graph[u][v] != 0 && dist[u] != INT_MAX && dist[ u] = dist[ u] + graph[u][v] )
{
dist[v] = dist[v] + graph[u][v];
}}}
//Display shortest distance
printf("Vertex\tDistance from source 0\n");
for (i = 0; i < V; i++)
printf("%d\t %d\n", i, dist[i]);
return 0;
}

ip fragmentation

#include stdio.h
#define HEADER 20
int main()
{
int data, mtu, id;
int size, i;
int fragment = 1;
int total = 0;
printf("Enter IP data size:");
scanf("%d", &data);
printf("Enter MTU Size:");
scanf("%d", &mtu);
printf("Enter Identification ID:");
scanf("%d", &id);
size = mtu - HEADER;
printf("\n Data per fragment = %d bytes \n", size);
printf("\n ----- IP Fragmentation ----- \n");
printf("|Fragment|Identification|Data Bytes|MF|\n");
printf("+--------+--------------+----------+---+\n");
for (i = 0; i < data; i = i + size)
{
int current, mf;
if (i + size <= data)
current = size;
else
current = data - i;
if (i + size < data)
mf = 1;
else
mf = 0;
printf("|%-8d|%-14d|%-10d|%-8d|\n",
fragment, id, current, mf);
printf("+--------+--------------+----------+---+\n");
total = total + current;
fragment++;
}
printf("\n Original IP Data: %d bytes", data);
printf("\n Reassembled IP Data: %d bytes", total);
if (data == total)
printf("\n\n Reassembly Successful");
else
printf("\n\n Reassembly Failed");
return 0;
}
